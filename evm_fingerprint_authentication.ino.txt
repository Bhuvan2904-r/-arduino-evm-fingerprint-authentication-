#include <Adafruit_Fingerprint.h>
#include <LiquidCrystal.h>
#include <SoftwareSerial.h>
#include <EEPROM.h>

// --------------------------------------------------
// Fingerprint Sensor
// R307 TX -> Arduino D2
// R307 RX -> Arduino D3
// --------------------------------------------------
SoftwareSerial fingerprintSerial(2, 3);
Adafruit_Fingerprint finger = Adafruit_Fingerprint(&fingerprintSerial);

// --------------------------------------------------
// LCD
// RS, EN, D4, D5, D6, D7
// --------------------------------------------------
LiquidCrystal lcd(7, 8, 9, 10, 11, 12);

// --------------------------------------------------
// Candidate Voting Buttons
// --------------------------------------------------
const int candidate1Button = 4;
const int candidate2Button = 5;
const int candidate3Button = 6;

// --------------------------------------------------
// Buzzer
// --------------------------------------------------
const int buzzerPin = 13;

// --------------------------------------------------
// Vote Storage
// EEPROM addresses
// --------------------------------------------------
const int candidate1Address = 0;
const int candidate2Address = 1;
const int candidate3Address = 2;

// --------------------------------------------------
// Voting Status
// --------------------------------------------------
bool voterAuthenticated = false;
bool voterHasVoted = false;

int candidate1Votes = 0;
int candidate2Votes = 0;
int candidate3Votes = 0;

// --------------------------------------------------
// Function Declarations
// --------------------------------------------------
void displayMessage(const char *line1, const char *line2);
bool authenticateFingerprint();
void castVote(int candidate);
void loadVotes();
void saveVotes();
void beep();
void showResults();

// --------------------------------------------------
// SETUP
// --------------------------------------------------
void setup()
{
    Serial.begin(9600);

    fingerprintSerial.begin(57600);

    lcd.begin(16, 2);

    pinMode(candidate1Button, INPUT_PULLUP);
    pinMode(candidate2Button, INPUT_PULLUP);
    pinMode(candidate3Button, INPUT_PULLUP);

    pinMode(buzzerPin, OUTPUT);

    loadVotes();

    lcd.clear();
    lcd.setCursor(0, 0);
    lcd.print("Biometric EVM");
    lcd.setCursor(0, 1);
    lcd.print("Initializing...");

    delay(2000);

    finger.begin(57600);

    if (finger.verifyPassword())
    {
        Serial.println("Fingerprint sensor detected.");
        displayMessage("System Ready", "Place Finger");
    }
    else
    {
        Serial.println("Fingerprint sensor not detected.");
        displayMessage("Sensor Error", "Check Wiring");
    }

    delay(2000);
}

// --------------------------------------------------
// MAIN LOOP
// --------------------------------------------------
void loop()
{
    voterAuthenticated = false;
    voterHasVoted = false;

    displayMessage("Place Finger", "To Authenticate");

    if (authenticateFingerprint())
    {
        voterAuthenticated = true;

        beep();

        displayMessage("Access Granted", "Select Candidate");

        delay(1000);

        // Candidate 1
        if (digitalRead(candidate1Button) == LOW)
        {
            castVote(1);
        }

        // Candidate 2
        else if (digitalRead(candidate2Button) == LOW)
        {
            castVote(2);
        }

        // Candidate 3
        else if (digitalRead(candidate3Button) == LOW)
        {
            castVote(3);
        }
    }

    delay(500);
}

// --------------------------------------------------
// FINGERPRINT AUTHENTICATION
// --------------------------------------------------
bool authenticateFingerprint()
{
    uint8_t result;

    result = finger.getImage();

    if (result != FINGERPRINT_OK)
        return false;

    result = finger.image2Tz();

    if (result != FINGERPRINT_OK)
        return false;

    result = finger.fingerFastSearch();

    if (result == FINGERPRINT_OK)
    {
        Serial.print("Fingerprint matched. ID: ");
        Serial.println(finger.fingerID);

        return true;
    }

    displayMessage("Access Denied", "Unknown Finger");

    Serial.println("Fingerprint not recognized.");

    delay(1500);

    return false;
}

// --------------------------------------------------
// CAST VOTE
// --------------------------------------------------
void castVote(int candidate)
{
    if (!voterAuthenticated)
    {
        displayMessage("Access Denied", "Authenticate");
        delay(1500);
        return;
    }

    if (voterHasVoted)
    {
        displayMessage("Already Voted", "Access Denied");
        delay(1500);
        return;
    }

    switch (candidate)
    {
        case 1:
            candidate1Votes++;
            break;

        case 2:
            candidate2Votes++;
            break;

        case 3:
            candidate3Votes++;
            break;
    }

    saveVotes();

    voterHasVoted = true;

    beep();

    displayMessage("Vote Recorded", "Successfully");

    Serial.print("Vote recorded for Candidate ");
    Serial.println(candidate);

    delay(2000);
}

// --------------------------------------------------
// LOAD VOTES FROM EEPROM
// --------------------------------------------------
void loadVotes()
{
    candidate1Votes = EEPROM.read(candidate1Address);
    candidate2Votes = EEPROM.read(candidate2Address);
    candidate3Votes = EEPROM.read(candidate3Address);
}

// --------------------------------------------------
// SAVE VOTES TO EEPROM
// --------------------------------------------------
void saveVotes()
{
    EEPROM.update(candidate1Address, candidate1Votes);
    EEPROM.update(candidate2Address, candidate2Votes);
    EEPROM.update(candidate3Address, candidate3Votes);
}

// --------------------------------------------------
// BUZZER
// --------------------------------------------------
void beep()
{
    digitalWrite(buzzerPin, HIGH);
    delay(150);
    digitalWrite(buzzerPin, LOW);
}

// --------------------------------------------------
// LCD MESSAGE
// --------------------------------------------------
void displayMessage(const char *line1, const char *line2)
{
    lcd.clear();

    lcd.setCursor(0, 0);
    lcd.print(line1);

    lcd.setCursor(0, 1);
    lcd.print(line2);
}

// --------------------------------------------------
// DISPLAY RESULTS
// --------------------------------------------------
void showResults()
{
    Serial.println("--------------------");
    Serial.println("Voting Results");
    Serial.print("Candidate 1: ");
    Serial.println(candidate1Votes);

    Serial.print("Candidate 2: ");
    Serial.println(candidate2Votes);

    Serial.print("Candidate 3: ");
    Serial.println(candidate3Votes);

    Serial.println("--------------------");
}
